# Explicit bench operations. No action runs merely by sourcing without arguments.
# args: server action query_id session_id [duration_seconds]
if {[llength $argv]<4} {error {Need server action query_id session_id}}
set server [lindex $argv 0]
set action [lindex $argv 1]
set qid [string tolower [lindex $argv 2]]
set sid [string tolower [lindex $argv 3]]
if {$action ne "run"} {error {This reviewed entry supports run only}}
if {![regexp {^[0-9a-f]{32}$} $qid] || $qid eq [string repeat 0 32]} {error {Fresh nonzero query ID required}}
if {![regexp {^[0-9a-f]{16}$} $sid] || $sid eq [string repeat 0 16]} {error {Nonzero session ID required}}
set root [file dirname [file normalize [info script]]]
set seconds 5
if {[llength $argv]>4} {set seconds [lindex $argv 4]}
if {![string is integer -strict $seconds] || $seconds<1 || $seconds>10} {error {Bench run must be 1..10 seconds}}
set mf [open $root/core-uuid.txt r];set expected_uuid [string trim [read $mf]];close $mf
proc sample {} {
 global v status ctl
 refresh_hw_vio -update_output_values $v
 set raw [get_property INPUT_VALUE $status]
 scan $raw %x value
 puts "VIO_STATUS=$raw CONTROL=[get_property OUTPUT_VALUE $ctl]"
 return $value
}
proc controls {value} {
 global v ctl
 set_property OUTPUT_VALUE $value $ctl
 commit_hw_vio $v
 # Do not insert slow remote refresh between pulse edges. Final state is sampled separately.
}
proc pulse {value} { controls $value; after 20; controls 02 }
proc wait_state {wanted} {
 for {set i 0} {$i<25} {incr i} {
  set s [sample]
  if {$s&32} {error {FPGA fault latched}}
  if {($s&15)==$wanted} {return $s}
  after 20
 }
 error {State wait expired}
}
open_hw_manager
connect_hw_server -url $server
set ts [get_hw_targets -quiet *210512180081*]
if {[llength $ts]!=1} {error {Expected unique HS1 serial 210512180081}}
current_hw_target [lindex $ts 0]
set_property PARAM.FREQUENCY 1000000 [current_hw_target]
open_hw_target
set motion_attempted 0
try {
 set ds [get_hw_devices -quiet xc7z100*]
 if {[llength $ds]!=1} {error {Expected unique xc7z100}}
 set d [lindex $ds 0];current_hw_device $d
 set_property PROBES.FILE $root/c1_control.ltx $d
 if {$action eq "program"} {
  set_property PROGRAM.FILE $root/c1_control.bit $d
  program_hw_devices $d
 }
 refresh_hw_device $d
 if {[get_property REGISTER.CONFIG_STATUS.BIT14_DONE_PIN $d]!=1 || [get_property {REGISTER.CONFIG_STATUS.BIT04_END_OF_STARTUP_(EOS)_STATUS} $d]!=1} {error {DONE/EOS not asserted}}
 set identity [string tolower [get_property REGISTER.USR_ACCESS $d]]
 if {[string trimleft $identity 0x] ne "c103f401"} {error {Wrong control build identity; do not use VIO UUID alone}}
 set vs [get_hw_vios -quiet -of_objects $d]
 if {[llength $vs]!=1} {error {Expected one VIO}}
 set v [lindex $vs 0]
 if {[string toupper [get_property CORE_UUID $v]] ne [string toupper $expected_uuid]} {error {BIT/LTX VIO UUID mismatch}}
 set ctl [get_hw_probes c1_controls -of_objects $v]
 set status [get_hw_probes c1_status -of_objects $v]
 set qry [get_hw_probes c1_query_id -of_objects $v]
 set sess [get_hw_probes c1_logging_session -of_objects $v]
 set s [sample]
 if {$action eq "program"} {
  if {[get_property OUTPUT_VALUE $ctl] ne "00" || ($s&1023)!=0} {error {Unexpected power-on control/status}}
  after 30000
  refresh_hw_device $d
  if {[get_property REGISTER.CONFIG_STATUS.BIT14_DONE_PIN $d]!=1 || [get_property {REGISTER.CONFIG_STATUS.BIT04_END_OF_STARTUP_(EOS)_STATUS} $d]!=1} {error {Delayed DONE/EOS check failed}}
  puts CONTROL_BIT_PROGRAMMED_DONE_EOS_CONFIRMED_NO_ARM
 } elseif {$action eq "query"} {
  if {($s&15)!=0 && ($s&15)!=6} {error {Query requires stopped/idle or reviewed fault}}
  controls 02;after 100
  if {!([sample]&16)} {error {UART ownership not acquired}}
  set_property OUTPUT_VALUE $qid $qry;commit_hw_vio $v
  pulse 06;wait_state 3
  puts "QUERY_SENT=$qid; independently inspect S1SYNC CRC and nonce"
 } elseif {$action eq "stop"} {
  pulse 12;after 300
  if {!([sample]&64)} {error {STOP not confirmed}}
  puts STOP_CONFIRMED
 } else {
  if {[string tolower [get_property OUTPUT_VALUE $qry]] ne $qid || [get_property OUTPUT_VALUE $ctl] ne "02"} {error {Query/control identity changed}}
  if {$action eq "bind"} {
   if {($s&15)!=3 || ($s&32) || ($s&768)} {error {Binding requires query wait and PWM off}}
   set_property OUTPUT_VALUE $sid $sess;commit_hw_vio $v
   pulse 0a;after 300
   if {!([sample]&128)} {error {Measurement session did not bind}}
   puts "SESSION_BOUND=$sid; independently inspect V2A and F1 CRC"
  } else {
   if {[string tolower [get_property OUTPUT_VALUE $sess]] ne $sid || ($s&1023)!=211} {error {Run requires bound query wait, no fault, PWM off}}
   # Caller must first verify current firmware, fresh F1, and the physical bench.
   puts PRE_MOTION_READY_WAITING_FOR_LOCAL_RELEASE
   set gate [file join $root run-release.txt]
   set gate_deadline [expr {[clock milliseconds]+180000}]
   while {![file exists $gate]} {
    if {[clock milliseconds]>$gate_deadline} {error {No current observation release; no ARM sent}}
    after 500
   }
   exec {D:/c1-recovery-20261103/pyocd-host/Scripts/python.exe} D:/c1-recovery-20261103/check-live-f1.py
   set motion_attempted 1
   pulse 22;wait_state 4
   pulse 42
   # FPGA enforces its independent 30-second authorization deadline.
   after [expr {1000*$seconds}]
   pulse 12;after 300
   if {!([sample]&64)} {error {STOP not confirmed}}
   set motion_attempted 0
   puts BENCH_WINDOW_ENDED_STOP_CONFIRMED_CHECK_UART_AUTO_ACK_AND_PHYSICAL_VIDEO
  }
 }
} finally {
 if {$motion_attempted} {catch {pulse 12};after 300;catch {sample}}
 close_hw_target
 disconnect_hw_server
 close_hw_manager
}
exit
