"""Receive bench evidence from PC2 on the private VPN; never executes uploads."""
from http.server import BaseHTTPRequestHandler,ThreadingHTTPServer
from pathlib import Path
import hashlib,json,uuid,datetime
OUT=Path('D:/c1-recovery-20261103/pc2-result-inbox');OUT.mkdir(exist_ok=True)
class Handler(BaseHTTPRequestHandler):
 def do_POST(self):
  if self.client_address[0] not in ['192.0.2.2','192.0.2.1'] or self.path!='/result':
   self.send_error(403);return
  try:n=int(self.headers.get('Content-Length','0'))
  except ValueError:self.send_error(400);return
  if not 0<n<=8*1024*1024:self.send_error(413);return
  self.connection.settimeout(30)
  data=self.rfile.read(n)
  if len(data)!=n:self.send_error(400);return
  stamp=datetime.datetime.now().strftime('%Y%m%d-%H%M%S')+'-'+uuid.uuid4().hex[:8]
  p=OUT/(stamp+'.zip');p.write_bytes(data)
  receipt={'file':p.name,'bytes':n,'sha256':hashlib.sha256(data).hexdigest(),'source':self.client_address[0]}
  (OUT/(stamp+'.json')).write_text(json.dumps(receipt,indent=2))
  raw=json.dumps(receipt).encode();self.send_response(200);self.send_header('Content-Length',str(len(raw)));self.end_headers();self.wfile.write(raw)
  print(json.dumps(receipt),flush=True)
ThreadingHTTPServer(('192.0.2.1',18766),Handler).serve_forever()
