import sys,json,hashlib,time
from pathlib import Path
sys.path.insert(0,r'C:/Users/USER/.codex/skills/fpga-remote-lab/scripts')
from remote_lab import TCF
root=Path('D:/c1-recovery-20261103/recovery-package')
p='D:/FPGA_Remote_JTAG/c1-recovery-20261103/local-download-'+time.strftime('%H%M%S')
receipt={'remote_directory':p,'verified':False,'files':{}}
out=Path('D:/c1-recovery-20261103/local-download-receipt.json')

c=TCF('192.0.2.2',timeout=10,bind_ip='192.0.2.1')
