# Finite observation-only valid frames: no motion, PWM, optical or filesystem writes.
import time,ustruct
from pyb import UART
uart=UART(3,115200)
frame=ustruct.pack('<BBhhBB',0xaa,0xff,0,0,0,0xee)
start=time.ticks_ms()
count=0
while time.ticks_diff(time.ticks_ms(),start)<20000:
    uart.write(frame)
    count+=1
    time.sleep_ms(2)
print('UART_VALID_PROBE_DONE',count)
