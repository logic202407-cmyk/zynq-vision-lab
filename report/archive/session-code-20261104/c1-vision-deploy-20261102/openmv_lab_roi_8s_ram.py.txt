# Finite image-only RAM diagnostic; no UART, PWM or filesystem writes.
import sensor,time
sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QQVGA)
sensor.skip_frames(time=2000)
sensor.set_auto_whitebal(False)
sensor.set_auto_gain(False)
roi=(106,18,12,12)
start=time.ticks_ms()
last=start
while time.ticks_diff(time.ticks_ms(),start)<8000:
    img=sensor.snapshot()
    stats=img.get_statistics(roi=roi)
    img.draw_rectangle(roi,color=(0,255,0))
    now=time.ticks_ms()
    if time.ticks_diff(now,last)>=2000:
        print('ROI_LAB',stats.l_min,stats.l_lq,stats.l_median,stats.l_uq,stats.l_max,stats.a_min,stats.a_lq,stats.a_median,stats.a_uq,stats.a_max,stats.b_min,stats.b_lq,stats.b_median,stats.b_uq,stats.b_max)
        last=now
print('ROI_DIAGNOSTIC_DONE')
