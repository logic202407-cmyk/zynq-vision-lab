def clean(t):
    t=re.sub(r'^\| Host\s*:.*$', '| Host : REDACTED',t,flags=re.M)
    t=t.replace('LAPTOP-TU4U942T','REDACTED_HOST')
    t=re.sub(r'[A-Za-z]:[\\/]Users[\\/][^\\/\s]+', '<USER>',t)
    return t
