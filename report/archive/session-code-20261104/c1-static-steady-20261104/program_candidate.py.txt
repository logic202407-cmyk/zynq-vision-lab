PROBE='REDACTED_DEVICE_ID'

p.add_argument('--output',type=Path,required=True)
a=p.parse_args()

a.output.mkdir(parents=True,exist_ok=False)
logging.basicConfig(level=logging.INFO,handlers=[logging.FileHandler(a.output/'backend.log'),logging.StreamHandler()])

session=None
def save(): (a.output/'result.json').write_text(json.dumps(result,indent=2))
def require(value,message):
 if not value:raise RuntimeError(message)
def read_full(t):
 data=bytearray()
 for offset in range(0,524288,4096):
  chunk=bytes(t.read_memory_block8(0x08000000+offset,4096))
  require(len(chunk)==4096,'Short Flash read')
  data.extend(chunk)
 return bytes(data)
try:
 binary=(ROOT/'build/skystar_tracking.bin').read_bytes()
 pack=Path('R:/evidence/c1-safe-deploy-20261101/Keil.STM32F4xx_DFP.3.1.1.pack')
 require(len(binary)==12356 and digest(binary)==BIN_HASH,'Candidate mismatch')
 require(digest(pack.read_bytes())==PACK_HASH,'Pack mismatch')

          'resume_on_disconnect':False,'cache.enable_memory':False,
          'cache.enable_register':False,'cache.read_code_from_elf':False,
          'chip_erase':'sector','smart_flash':False,'fast_program':False,
          'enable_swv':False}

except Exception:
