# RAM-only image diagnostic: no UART, PWM, laser or filesystem writes.
# Reuses repository sensor format and red LAB threshold; not a tracking test.
import sensor, time
sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QQVGA)
sensor.skip_frames(time=2000)
sensor.set_auto_whitebal(False)
sensor.set_auto_gain(False)
threshold = (32, 43, 13, 51, 4, 52)
start = time.ticks_ms()
last = start
frames = 0
while time.ticks_diff(time.ticks_ms(), start) < 30000:
    img = sensor.snapshot()
    blobs = [b for b in img.find_blobs([threshold], pixels_threshold=8, area_threshold=8, merge=True) if b.area() >= 20]
    best = max(blobs, key=lambda b: b.pixels()) if blobs else None
    if best:
        img.draw_rectangle(best.rect(), color=(0,255,0))
        img.draw_cross(best.cx(), best.cy(), color=(0,255,0))
    frames += 1
    now = time.ticks_ms()
    if time.ticks_diff(now, last) >= 500:
        print('IMAGE_ONLY', frames, 'candidates', len(blobs), 'target', (best.cx(),best.cy(),best.pixels()) if best else None)
        last = now
print('IMAGE_DIAGNOSTIC_DONE', frames)
