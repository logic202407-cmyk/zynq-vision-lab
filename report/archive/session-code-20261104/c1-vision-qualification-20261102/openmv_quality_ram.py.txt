# Finite image-only RAM diagnostic. No UART, actuator or filesystem writes.
"""Single-target observation policy. No sensor, UART, PWM or filesystem access.

The caller supplies already thresholded blobs. This is a quality gate, not
cross-camera identity or mechanical calibration. One instance per camera.
"""


def blob_value(blob, name):
    value = getattr(blob, name)
    return value() if callable(value) else value


class RedTargetPolicy:
    def __init__(self, min_area=20, border=1, stable_frames=3):
        if min_area < 1 or border < 1 or stable_frames < 1:
            raise ValueError('Positive observation policy required')
        self.min_area = min_area
        self.border = border
        self.stable_frames = stable_frames
        self.good_frames = 0
        self.reason = 'MISSING'

    def choose(self, blobs, width=160, height=120):
        candidates = [b for b in blobs if blob_value(b, 'area') >= self.min_area]
        reason = 'MISSING' if not candidates else 'AMBIGUOUS' if len(candidates) != 1 else None
        blob = candidates[0] if reason is None else None
        if blob is not None:
            x, y, w, h = blob_value(blob, 'rect')
            cx, cy = blob_value(blob, 'cx'), blob_value(blob, 'cy')
            if (w <= 0 or h <= 0 or x < 0 or y < 0 or x+w > width or y+h > height or
                    not x <= cx < x+w or not y <= cy < y+h):
                reason = 'GEOMETRY'
            elif (x < self.border or y < self.border or
                  x+w > width-self.border or y+h > height-self.border):
                reason = 'EDGE'
        if reason is not None:
            self.good_frames = 0
            self.reason = reason
            return None
        self.good_frames = min(self.good_frames+1, self.stable_frames)
        self.reason = 'VALID' if self.good_frames == self.stable_frames else 'WARMUP'
        return blob if self.reason == 'VALID' else None


def target_offsets(blob, width=160, height=120):
    if blob is None:
        return 0, 0, 0
    return blob_value(blob, 'cx')-width//2, height//2-blob_value(blob, 'cy'), 1


import sensor
import time
sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QQVGA)
sensor.skip_frames(time=2000)
sensor.set_auto_whitebal(False)
sensor.set_auto_gain(False)
PROFILES = [('repository_default', (32, 43, 13, 51, 4, 52)), ('explicit_bench_reference', (10, 80, 20, 90, -10, 80))]
CONFIG = {'red_threshold': (32, 43, 13, 51, 4, 52), 'PIXELS_THRESHOLD': 8, 'AREA_THRESHOLD': 8, 'MIN_BLOB_AREA': 20, 'FRAME_BORDER': 1, 'STABLE_FRAMES': 3}
policies = [RedTargetPolicy(CONFIG['MIN_BLOB_AREA'], CONFIG['FRAME_BORDER'], CONFIG['STABLE_FRAMES']) for _ in PROFILES]
for name in ('get_hmirror', 'get_vflip', 'get_transpose'):
    getter = getattr(sensor, name, None)
    print('CAMERA_GEOMETRY', name, getter() if callable(getter) else 'UNKNOWN')
print('QUALITY_CONFIG', PROFILES, CONFIG, 'NO_UART_NO_FLASH_NO_ACTUATORS')
started = time.ticks_ms()
last = started
while time.ticks_diff(time.ticks_ms(), started) < 30000:
    image = sensor.snapshot()
    results = []
    # Both detections run before drawing so overlay pixels never contaminate the second profile.
    for index, item in enumerate(PROFILES):
        blobs = image.find_blobs([item[1]], pixels_threshold=CONFIG['PIXELS_THRESHOLD'], area_threshold=CONFIG['AREA_THRESHOLD'], merge=False)
        selected = policies[index].choose(blobs, 160, 120)
        candidates = sum(1 for b in blobs if blob_value(b, 'area') >= CONFIG['MIN_BLOB_AREA'])
        results.append((selected, candidates, target_offsets(selected)))
    chosen = results[-1][0]
    if chosen is not None:
        image.draw_rectangle(blob_value(chosen, 'rect'), color=(0, 255, 0))
        image.draw_cross((blob_value(chosen, 'cx'), blob_value(chosen, 'cy')), color=(0, 255, 0))
    image.draw_string((2, 2), PROFILES[-1][0], color=(255, 255, 0))
    image.draw_string((2, 14), policies[-1].reason, color=(255, 255, 0))
    now = time.ticks_ms()
    if time.ticks_diff(now, last) >= 500:
        for index, item in enumerate(PROFILES):
            print('QUALITY_IMAGE_ONLY', item[0], policies[index].reason, results[index][1], results[index][2])
        last = now
print('QUALITY_IMAGE_DIAGNOSTIC_DONE')
