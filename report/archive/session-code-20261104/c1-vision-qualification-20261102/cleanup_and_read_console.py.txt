import ctypes,os,runpy,time
from ctypes import wintypes
from PIL import ImageGrab
u=ctypes.windll.user32;u.SetProcessDPIAware();u.GetForegroundWindow.restype=wintypes.HWND
windows=[]
@ctypes.WINFUNCTYPE(wintypes.BOOL,wintypes.HWND,wintypes.LPARAM)
def visit(h,_):
    if u.IsWindowVisible(h):
        pid=wintypes.DWORD();u.GetWindowThreadProcessId(h,ctypes.byref(pid))
        if pid.value==42692:
            name=ctypes.create_unicode_buffer(512);u.GetWindowTextW(h,name,512)
            windows.append((h,name.value))
    return True
u.EnumWindows(visit,0)
for h,name in windows:
    if name=='OpenMV IDE':u.PostMessageW(h,0x0010,0,0)
os.environ['OPENMV_TARGET_TITLE']='openmv_quality_ram.py - OpenMV IDE'
runpy.run_path('R:/evidence/c1-vision-qualification-20261102/focus_ide.py')
h=u.GetForegroundWindow();title=ctypes.create_unicode_buffer(512);u.GetWindowTextW(h,title,512)
if title.value!='openmv_quality_ram.py - OpenMV IDE':raise RuntimeError('No scrolling in another window')
u.SetCursorPos(1200,1300)
u.mouse_event(0x0800,0,0,4800,0) # Scroll the existing log, no keyboard/device input.
time.sleep(.5)
if u.GetForegroundWindow()!=h:raise RuntimeError('Foreground changed; no screenshot')
rect=wintypes.RECT();u.GetWindowRect(h,ctypes.byref(rect))
ImageGrab.grab(bbox=(rect.left,rect.top,rect.right,rect.bottom)).save('R:/evidence/c1-vision-qualification-20261102/quality-console-start-private.png')
print('READ_EXISTING_IDE_CONSOLE_ONLY')
