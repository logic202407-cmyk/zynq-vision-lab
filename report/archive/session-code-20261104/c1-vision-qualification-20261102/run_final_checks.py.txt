import json,os,subprocess,sys,time
from pathlib import Path
root=Path('R:/clock-repo');stage=Path(__file__).parent
env=os.environ.copy();env.update(TEMP=str(stage/'temp'),TMP=str(stage/'temp'),PYTHONDONTWRITEBYTECODE='1',PYTHONUTF8='1')
commands=[('python-final',[sys.executable,'-X','utf8','-m','unittest','discover','-s','tests','-v']),
          ('repository-final',[sys.executable,'-X','utf8','tools/check_repository.py'])]
records=[]
for name,command in commands:
    start=time.monotonic();r=subprocess.run(command,cwd=root,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    (stage/(name+'.txt')).write_bytes(r.stdout)
    records.append(dict(command=command,exit_code=r.returncode,seconds=time.monotonic()-start,log=name+'.txt'))
    print(json.dumps(records[-1]),flush=True)
    if r.returncode:print(r.stdout.decode('utf-8','replace')[-3000:],flush=True)
(stage/'final-checks.json').write_text(json.dumps(dict(python=sys.version,temp=env['TEMP'],commands=records),indent=2),encoding='utf-8')
raise SystemExit(any(r['exit_code'] for r in records))
