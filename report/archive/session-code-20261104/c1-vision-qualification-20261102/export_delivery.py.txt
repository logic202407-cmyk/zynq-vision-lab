import hashlib,json,subprocess
from pathlib import Path
stage=Path(__file__).parent;root=Path('R:/clock-repo')
out=root/'report/evidence/c1-vision-qualification-20261102';out.mkdir(exist_ok=True)
def read(path):return json.loads(path.read_text(encoding='utf-8'))
def write(name,data):(out/name).write_bytes((json.dumps(data,ensure_ascii=False,indent=2)+'\n').encode('utf-8'))
def write_wire(name,data):
    fields=[]
    for key,value in data.items():
        if isinstance(value,list):
            body='[\n'+',\n'.join('    '+json.dumps(item,ensure_ascii=False,separators=(',',':')) for item in value)+'\n  ]'
        else:body=json.dumps(value,ensure_ascii=False)
        fields.append('  '+json.dumps(key)+': '+body)
    text='{\n'+',\n'.join(fields)+'\n}\n'
    assert json.loads(text)==data
    (out/name).write_bytes(text.encode('utf-8'))
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def log(source,name):
    text=source.read_text(encoding='utf-8').replace('\r','')
    text='\n'.join(line.rstrip() for line in text.splitlines()).rstrip('\n')+'\n'
    (out/name).write_bytes(text.encode('utf-8'))
