def read(path):return json.loads(path.read_text(encoding='utf-8'))
def write(name,value):(out/name).write_bytes((json.dumps(value,ensure_ascii=False,indent=2)+'\n').encode('utf-8'))
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def text_copy(source,name):
    text=source.read_text(encoding='utf-8').replace('\r','').rstrip('\n')+'\n'
    text=re.sub(r'(?m)^\| Host\s*:.*$', '| Host         : <private-host>',text)
    text='\n'.join(line.rstrip() for line in text.splitlines()).rstrip('\n')+'\n'
    (out/name).write_bytes(text.encode('utf-8'))
